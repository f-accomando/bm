# bm - Raspberry Pi Zero / Zero W boot configuration

kernel=kernel.img
arm_64bit=0

# Always drive HDMI even if no monitor is detected at power-on
hdmi_force_hotplug=1
# HDMI (not DVI) mode, so the TV also gets audio/proper levels
hdmi_drive=2
# Use the full screen area; our pattern draws a 1px border to check this
disable_overscan=1

# The kernel asks for a 640x360 framebuffer and the GPU scales it to the
# HDMI mode; nearest-neighbour keeps the text sharp instead of blurred.
scaling_kernel=8

# Always 1080p60 (16:9, CEA): the picture keeps its shape. Left to the
# monitor's list, an ultrawide one may be driven at its 21:9 mode and the
# 16:9 picture gets stretched. If the monitor still fills the width, set
# its aspect option to "original" / "1:1". For 720p60: hdmi_mode=4.
hdmi_group=1
hdmi_mode=16

# Skip the rainbow splash
disable_splash=1

# Serial console: the kernel itself routes PL011 (UART0) to GPIO14/15 at
# 115200 8N1 and reads the UART clock from the firmware, so nothing else is
# needed here. enable_uart=1 only keeps the core clock stable.
enable_uart=1
