// ===========================================================================
// NES game in bm C (NROM). It builds as it is: a blue screen and a ball that
// the pad moves. Ctrl+B builds, Ctrl+R runs, F6 checks, F7 shows the assembly.
// Sections marked //@name ... //@end are where the Sprite, Sounds and Map tabs
// read and write their data. Keep the markers.
// ===========================================================================

byte x = 124;              // where the ball is on the screen
byte y = 112;

void start(void)           // once, when the game starts
{
    background(BLUE);
    sprite_palette(0, WHITE, SKY, NAVY);
}

void update(void)          // once every frame (60 times a second)
{
    // held(button): the button is down. pressed(button): it went down now.
    // The buttons: A B SELECT START UP DOWN LEFT RIGHT
    if (held(LEFT))  x -= 1;
    if (held(RIGHT)) x += 1;
    if (held(UP))    y -= 1;
    if (held(DOWN))  y += 1;
    if (pressed(A)) play_sfx(sfx_jump);     // an effect of //@music (F5)
    sprite(0, x, y, 1);    // sprite 0 shows the tile 1 of the sheet
}

//@sprites
// Sprite sheet: the Sprite tab writes its tiles here, a tiles { } block between
// //@tiles and //@tiles-end (8x8 tiles, 2 bits a pixel, 16 bytes a tile, 16
// tiles in a row). The background takes the tiles from $100: leave $100 blank.
//@tiles  from the Sprite tab: 2bpp, 16 bytes a tile, the CHR ROM
tiles {
    blank(1), // $000
    0x3C,0x72,0xE1,0xC1,0x81,0x83,0x46,0x3C,0x3C,0x4E,0x9F,0xBF,0xFF,0xFF,0x7E,0x3C, // $001
}
//@tiles-end
//@end

//@music  from the Sounds tab: C1..B7 (CS4 sharp), __ nothing, OFF release
const byte sfx_jump[] = {          // sound effect
    2, PU1, DUTY50,                          // frames a step, channel, wave
    C4,15,__,  E4,14,__,  G4,13,__,  C5,12,UP+3,
    C5,8,UP+3,  C5,4,UP+3,
    END };
//@end
